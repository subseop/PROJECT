- SLAES 데이터 삽입 로직
```
DECLARE
    v_base_price    NUMBER;  -- 기본 단가
    v_base_qty      NUMBER;  -- 기본 판매량
    v_season_factor NUMBER;  -- 월별 계절 요인
    v_final_qty     NUMBER;  -- 최종 판매량
    v_final_price   NUMBER;  -- 최종 단가
BEGIN
    -- merchant_product 테이블의 모든 상품(점포-상품 매핑)을 순회
    FOR mp IN (
        SELECT merchant_product_id, product_id 
        FROM merchant_product
        ORDER BY merchant_product_id
    ) LOOP
        
        -- time_month 테이블의 2024년 모든 월(월ID/월번호)을 순회
        FOR tm IN (
            SELECT month_id, month 
            FROM time_month
            WHERE year in (2024, 2025)
            ORDER BY month_id
        ) LOOP
            
            -----------------------------------------------------
            -- A. 기본 판매량 및 기본 단가 설정
            -----------------------------------------------------
            CASE mp.product_id
                WHEN 1 THEN  -- 바나나
                    v_base_price := 4500; 
                    v_base_qty   := 300;
                WHEN 2 THEN  -- 우유
                    v_base_price := 2800; 
                    v_base_qty   := 450;
                WHEN 3 THEN  -- 빵
                    v_base_price := 3500; 
                    v_base_qty   := 200;
                WHEN 4 THEN  -- 달걀
                    v_base_price := 7800; 
                    v_base_qty   := 250;
                WHEN 5 THEN  -- 아보카도
                    v_base_price := 11500; 
                    v_base_qty   := 80;
                ELSE         -- 기타 상품
                    v_base_price := 5000; 
                    v_base_qty   := 100;
            END CASE;

            -----------------------------------------------------
            -- B. 계절 요인 적용 (상품/계절별 판매 패턴)
            -----------------------------------------------------
            v_season_factor := 1.0;

            -- 겨울(12,1,2)에는 빵/달걀이 잘 팔림
            IF tm.month IN (12, 1, 2) AND mp.product_id IN (3, 4) THEN
                v_season_factor := 1.2;
            END IF;

            -- 봄/가을에는 바나나 판매 증가
            IF tm.month IN (3, 4, 5, 9, 10) AND mp.product_id = 1 THEN
                v_season_factor := 1.1;
            END IF;

            -----------------------------------------------------
            -- C. 난수(0.8~1.2)를 섞어 최종 구매량 및 단가 산정
            -----------------------------------------------------

            -- 판매량 = 기본량 × 계절요인 × 랜덤변동
            v_final_qty :=
                ROUND(v_base_qty * v_season_factor * DBMS_RANDOM.VALUE(0.8, 1.2));

            -- 단가 = 기본가 × 랜덤변동(±5%) → 10원 단위로 반올림
            v_final_price := v_base_price * DBMS_RANDOM.VALUE(0.95, 1.05);
            v_final_price := ROUND(v_final_price / 10) * 10;

            -----------------------------------------------------
            -- D. 결과를 sales 테이블에 삽입
            -----------------------------------------------------
            INSERT INTO sales (merchant_product_id, month_id, qty, unit_price)
            VALUES (mp.merchant_product_id, tm.month_id, v_final_qty, v_final_price);

        END LOOP;
    END LOOP;

    COMMIT;
END;
/
```
 1. 모든 merchant_product에 있는 상품들을 2024년의 모든 달에 대해 루프
 2. 모든 상품에 대해 기본 판매량과 기본 가격 설정
 3. 상품별 계절에 따른 가중치 설정
 4. 난수를 섞어 최종 판매량과 최종 가격 결정
 5. 결과를 sales 테이블에 삽입
